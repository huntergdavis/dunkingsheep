usage: dunkingsheep task read [-h] [--summary] [--notes N|all]
                              [--kind {evidence,decision,handoff,progress,correction,note}]
                              [--before-note ID]
                              id

positional arguments:
  id

options:
  -h, --help            show this help message and exit
  --summary             bounded body and latest note of each kind
  --notes N|all         latest N notes (CEILING 200), or all for a 200-note
                        history page, oldest first
  --kind {evidence,decision,handoff,progress,correction,note}
                        filter --notes by kind; note includes legacy events
  --before-note ID      continue --notes before this scoped note cursor
