                              [--kind {work,review,verify,report}]

  --status {open,in_progress,blocked,done,cancelled}
  --assignee PANE
  --all                 include done/cancelled
  --limit, -n LIMIT
  --kind {work,review,verify,report}
  --needs-owner         only tasks blocked on a 'needs-owner:' question
